var ErrTemplateHasNoReadyReplica = errors.New("no ready replica")

type ReplicaStatus struct{ NodeID, Status string }
type requestPolicyKey struct{}
type queryResult struct{ Error error }
type fakeDB struct {
	mu            sync.Mutex
	rows          map[string][]ReplicaStatus
	listReads     map[string]int
	singleReads   map[string]int
	health        map[string]bool
	checks        int
	failure       error
	blocked       <-chan struct{}
	entered       chan struct{}
	singleBlocked <-chan struct{}
	singleEntered chan struct{}
}
type fakeQuery struct {
	db                 *fakeDB
	ctx                context.Context
	templateID, nodeID string
}

var store struct{ db *fakeDB }

func isReady() bool                                            { return store.db != nil }
func replicaModelToStatus(replica ReplicaStatus) ReplicaStatus { return replica }
func (db *fakeDB) WithContext(ctx context.Context) *fakeQuery  { return &fakeQuery{db: db, ctx: ctx} }
func (q *fakeQuery) Table(_ string) *fakeQuery                 { return q }
func (q *fakeQuery) Where(_ string, templateID, nodeID string) *fakeQuery {
	q.templateID, q.nodeID = templateID, nodeID
	return q
}
func (q *fakeQuery) First(replica *ReplicaStatus) queryResult {
	q.db.mu.Lock()
	q.db.singleReads[q.templateID]++
	err := q.db.failure
	var snapshot ReplicaStatus
	found := false
	for _, row := range q.db.rows[q.templateID] {
		if row.NodeID == q.nodeID {
			snapshot, found = row, true
			break
		}
	}
	blocked, entered := q.db.singleBlocked, q.db.singleEntered
	q.db.mu.Unlock()
	if entered != nil {
		select {
		case entered <- struct{}{}:
		default:
		}
	}
	if blocked != nil {
		select {
		case <-blocked:
		case <-q.ctx.Done():
			return queryResult{q.ctx.Err()}
		}
	}
	if err != nil {
		return queryResult{err}
	}
	if !found {
		return queryResult{errors.New("not found")}
	}
	*replica = snapshot
	return queryResult{}
}
func ListReplicas(ctx context.Context, templateID string) ([]ReplicaStatus, error) {
	db := store.db
	db.mu.Lock()
	db.listReads[templateID]++
	rows := append([]ReplicaStatus(nil), db.rows[templateID]...)
	err, blocked, entered := db.failure, db.blocked, db.entered
	db.mu.Unlock()
	if entered != nil {
		select {
		case entered <- struct{}{}:
		default:
		}
	}
	if blocked != nil {
		select {
		case <-blocked:
		case <-ctx.Done():
			return nil, ctx.Err()
		}
	}
	return rows, err
}

// The real v0.7.1 predicate is preserved as Base. The injected live-health and
// context policy model target versions that extend that predicate; production
// v0.7.1 itself checks only trimmed READY status, not live node health.
func isReplicaSchedulableNow(ctx context.Context, replica ReplicaStatus) bool {
	db := store.db
	db.mu.Lock()
	db.checks++
	healthy := db.health[strings.TrimSpace(replica.NodeID)]
	db.mu.Unlock()
	allowed, hasPolicy := ctx.Value(requestPolicyKey{}).(bool)
	return isReplicaSchedulableNowBase(ctx, replica) && healthy && ctx.Err() == nil && (!hasPolicy || allowed)
}

		listReads: make(map[string]int), singleReads: make(map[string]int), health: map[string]bool{"a": true, "b": true}}
	store.db = db
	readyReplicaResultCache = newResolveTTLCache(resolveResultCacheTTL)
	replicaSchedulableResultCache = newResolveTTLCache(resolveResultCacheTTL)
	resolveResultFetchGroup = &templateFetchGroup{calls: make(map[string]*templateFetchCall)}
	return db
}

func TestInvalidationRefreshesBothMetadataCaches(t *testing.T) {
	db := setup()
	ctx := context.Background()
	db.rows["tpl-other"] = []ReplicaStatus{{"b", "READY"}}
	for _, templateID := range []string{"tpl", "tpl-other"} {
		if _, err := ResolveTemplateReadyReplica(ctx, templateID, ""); err != nil {
			t.Fatal(err)
		}
		if !isTemplateReplicaSchedulable(ctx, templateID, db.rows[templateID][0].NodeID) {
			t.Fatal("expected READY")
		}
	}
	db.rows["tpl"] = []ReplicaStatus{{"a", "FAILED"}}
	invalidateResolveResultCache("tpl")
	if _, err := ResolveTemplateReadyReplica(ctx, "tpl", ""); !errors.Is(err, ErrTemplateHasNoReadyReplica) {
		t.Fatal(err)
	}
	if isTemplateReplicaSchedulable(ctx, "tpl", "a") {
		t.Fatal("invalidated state must be reloaded")
	}
	if _, err := ResolveTemplateReadyReplica(ctx, "tpl-other", ""); err != nil {
		t.Fatal(err)
	}
	if !isTemplateReplicaSchedulable(ctx, "tpl-other", "b") {
		t.Fatal("unrelated key must survive")
	}
	if db.listReads["tpl"] != 2 || db.singleReads["tpl"] != 2 || db.listReads["tpl-other"] != 1 || db.singleReads["tpl-other"] != 1 {
		t.Fatalf("wrong invalidation scope: list=%v single=%v", db.listReads, db.singleReads)
	}
}

func TestConcurrentBurstSharesMetadataButNotDecision(t *testing.T) {
	db := setup()
	db.rows["tpl"] = []ReplicaStatus{{"a", "READY"}}
	release := make(chan struct{})
	db.blocked, db.entered = release, make(chan struct{}, 1)
	var wait sync.WaitGroup
	start := make(chan struct{})
	for i := 0; i < 100; i++ {
		wait.Add(1)
		go func(allowed bool) {
			defer wait.Done()
			<-start
			ctx := context.WithValue(context.Background(), requestPolicyKey{}, allowed)
			_, err := ResolveTemplateReadyReplica(ctx, "tpl", "a")
			if (err == nil) != allowed {
				t.Errorf("allowed=%v, error=%v", allowed, err)
			}
		}(i%2 == 0)
	}
	close(start)
	<-db.entered
	close(release)
	wait.Wait()
	if db.listReads["tpl"] != 1 || db.checks != 100 {
		t.Fatalf("read=%d checks=%d", db.listReads["tpl"], db.checks)
	}
}

// The barriers capture an old DB snapshot before invalidation. Timeouts are
// deadlock guards only; ordering is established by the entered/release channels.
func receive[T any](t *testing.T, channel <-chan T) T {
	t.Helper()
	select {
	case value := <-channel:
		return value
	case <-time.After(2 * time.Second):
		t.Fatal("query did not finish after the expected barrier")
		var zero T
		return zero
	}
}

type resolutionResult struct {
	replica ReplicaStatus
	err     error
}

func TestReadyInvalidationSeparatesFlightsAndRejectsOldRefill(t *testing.T) {
	db := setup()
	db.rows["tpl"] = []ReplicaStatus{{"a", "READY"}}
	release := make(chan struct{})
	var once sync.Once
	resume := func() { once.Do(func() { close(release) }) }
	db.blocked, db.entered = release, make(chan struct{}, 1)
	var workers sync.WaitGroup
	defer func() { resume(); workers.Wait() }()
	oldResult := make(chan resolutionResult, 1)
	workers.Add(1)
	go func() {
		defer workers.Done()
		replica, err := ResolveTemplateReadyReplica(context.Background(), "tpl", "")
		oldResult <- resolutionResult{replica, err}
	}()
	receive(t, db.entered)
	db.mu.Lock()
	db.rows["tpl"] = []ReplicaStatus{{"b", "READY"}}
	db.blocked, db.entered = nil, nil
	db.mu.Unlock()
	invalidateResolveResultCache("tpl")
	newResult := make(chan resolutionResult, 1)
	workers.Add(1)
	go func() {
		defer workers.Done()
		replica, err := ResolveTemplateReadyReplica(context.Background(), "tpl", "")
		newResult <- resolutionResult{replica, err}
	}()
	// A post-invalidation request must not join the still-blocked old flight.
	newer := receive(t, newResult)
	if newer.err != nil || newer.replica.NodeID != "b" {
		t.Fatalf("new generation: %v", newer)
	}
	resume()
	older := receive(t, oldResult)
	if older.err != nil || older.replica.NodeID != "b" {
		t.Fatalf("old reader reused obsolete metadata: %v", older)
	}
	if _, err := ResolveTemplateReadyReplica(context.Background(), "tpl", "b"); err != nil {
		t.Fatal(err)
	}
	if db.listReads["tpl"] != 2 {
		t.Fatalf("expected one read per generation, got %v", db.listReads)
	}
}

func TestSingleInvalidationSeparatesFlightsAndRejectsOldRefill(t *testing.T) {
	db := setup()
	db.rows["tpl"] = []ReplicaStatus{{"a", "READY"}}
	release := make(chan struct{})
	var once sync.Once
	resume := func() { once.Do(func() { close(release) }) }
	db.singleBlocked, db.singleEntered = release, make(chan struct{}, 1)
	var workers sync.WaitGroup
	defer func() { resume(); workers.Wait() }()
	oldResult := make(chan bool, 1)
	workers.Add(1)
	go func() {
		defer workers.Done()
		oldResult <- isTemplateReplicaSchedulable(context.Background(), "tpl", "a")
	}()
	receive(t, db.singleEntered)
	db.mu.Lock()
	db.rows["tpl"] = []ReplicaStatus{{"a", "FAILED"}}
	db.singleBlocked, db.singleEntered = nil, nil
	db.mu.Unlock()
	invalidateResolveResultCache("tpl")
	newResult := make(chan bool, 1)
	workers.Add(1)
	go func() {
		defer workers.Done()
		newResult <- isTemplateReplicaSchedulable(context.Background(), "tpl", "a")
	}()
	if receive(t, newResult) {
		t.Fatal("new generation selected failed replica")
	}
	resume()
	if receive(t, oldResult) {
		t.Fatal("old metadata overwrote the new generation")
	}
	if isTemplateReplicaSchedulable(context.Background(), "tpl", "a") {
		t.Fatal("new cache was polluted")
	}
	if db.singleReads["tpl"] != 2 {
		t.Fatalf("expected one read per generation, got %v", db.singleReads)
	}
}

	db := setup()
	db.rows["tpl"] = nil
	release := make(chan struct{})
	var once sync.Once
	resume := func() { once.Do(func() { close(release) }) }
	db.singleBlocked, db.singleEntered = release, make(chan struct{}, 1)
	var workers sync.WaitGroup
	defer func() { resume(); workers.Wait() }()
	oldResult := make(chan bool, 1)
	workers.Add(1)
	go func() {
		defer workers.Done()
		oldResult <- isTemplateReplicaSchedulable(context.Background(), "tpl", "a")
	}()
	receive(t, db.singleEntered)
	db.mu.Lock()
	db.rows["tpl"] = []ReplicaStatus{{"a", "READY"}}
	db.singleBlocked, db.singleEntered = nil, nil
	db.mu.Unlock()

func TestPendingReadHasNoValueTTLAndFailedSlotIsReleased(t *testing.T) {
	// An already-expired value TTL makes expiry deterministic without sleeps.
	// Pending reads still have to share their slot until they finish or fail.
	cache := newResolveTTLCache(-time.Second)
	_, hit, generation := cache.Lookup("tpl\x00ready")
	if hit {
		t.Fatal("new slot already has a value")
	}
	_, hit, pending := cache.Lookup("tpl\x00ready")
	if hit || pending != generation {
		t.Fatal("pending read expired before completing metadata I/O")
	}
	if !cache.Store("tpl\x00ready", generation, []ReplicaStatus{{"a", "READY"}}) {
		t.Fatal("pending query could not publish")
	}
	_, _, replacement := cache.Lookup("tpl\x00ready")
	if replacement == generation {
		t.Fatal("expired value reused a generation ID")
	}
	if cache.Store("tpl\x00ready", generation, true) || cache.Discard("tpl\x00ready", generation) {
		t.Fatal("obsolete generation modified replacement slot")
	}
	if !cache.Discard("tpl\x00ready", replacement) {
		t.Fatal("failed query did not release its placeholder")
	}
	_, _, next := cache.Lookup("tpl\x00ready")
	if next == replacement {
		t.Fatal("failed slot generation was recycled")
	}
}

func TestCanceledInvalidatedReadStopsRetryAndReleasesSlot(t *testing.T) {
	db := setup()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	release := make(chan struct{})
	db.blocked, db.entered = release, make(chan struct{}, 1)
	result := make(chan error, 1)
	var workers sync.WaitGroup
	workers.Add(1)
	defer func() { cancel(); workers.Wait() }()
	go func() {
		defer workers.Done()
		_, err := ResolveTemplateReadyReplica(ctx, "tpl", "a")
		result <- err
	}()
	receive(t, db.entered)
	invalidateResolveResultCache("tpl")
	cancel()
	if err := receive(t, result); !errors.Is(err, context.Canceled) {
		t.Fatalf("invalidated retry ignored cancellation: %v", err)
	}
	db.mu.Lock()
	db.blocked, db.entered = nil, nil
	db.mu.Unlock()
	if _, err := ResolveTemplateReadyReplica(context.Background(), "tpl", "a"); err != nil {
		t.Fatal(err)
	}
	if db.listReads["tpl"] != 2 {
		t.Fatalf("canceled read leaked a pending slot: %v", db.listReads)
	}
}

func TestObsoleteFlightRecheckDoesNotAllocateReplacement(t *testing.T) {
	cache := newResolveTTLCache(time.Second)
	_, _, generation := cache.Lookup("tpl\x00ready")
	cache.DeletePrefix("tpl\x00")
	if _, hit, current := cache.ReadCurrent("tpl\x00ready", generation); hit || current {
		t.Fatal("invalidated flight was treated as current")
	}
	cache.mu.Lock()
	remaining := len(cache.entries)
	cache.mu.Unlock()
	if remaining != 0 {
		t.Fatal("obsolete/canceled flight allocated an unowned pending slot")
	}
}
